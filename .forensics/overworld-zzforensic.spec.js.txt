import { test } from '@playwright/test';
import fs from 'node:fs';

const OUT = '.forensics';

function freshSave() {
  return {
    version: 1, grade: 3,
    party: [
      { id: 'knight-shadow', name: 'Shadow', hp: 52, maxHp: 52, xp: 0, level: 1 },
      { id: 'wizard-grandmage', name: 'Grand Mage', hp: 38, maxHp: 38, xp: 0, level: 1 },
      { id: 'bunny-pepper', name: 'Pepper', hp: 46, maxHp: 46, xp: 0, level: 1 },
    ],
    gold: 10, potions: 2,
    floors: Array.from({ length: 9 }, (_, i) => ({ id: i + 1, unlocked: i < 3, complete: false, bestStreak: 0 })),
    settings: { musicVolume: 0, sfxVolume: 0, reducedMotion: false },
    stats: { totalBattles: 0, totalCorrect: 0, totalWrong: 0, playTimeSec: 0, firstPlayedAt: Date.now(), lastPlayedAt: Date.now() },
  };
}

async function simTime(page) {
  return page.evaluate(() => window.__MW_OVERWORLD.stats().simTime);
}
async function waitSim(page, seconds) {
  const t0 = await simTime(page);
  await page.waitForFunction((t) => window.__MW_OVERWORLD.stats().simTime >= t, t0 + seconds, { timeout: 120_000 });
}
async function pos(page) {
  return page.evaluate(() => {
    const p = window.__MW_OVERWORLD._state.pos;
    return { x: p.x, y: p.y, z: p.z };
  });
}

test.beforeEach(async ({ context }) => {
  await context.route(/^https?:\/\/(?!127\.0\.0\.1|localhost)/, (r) => r.abort());
});

test('world forensics', async ({ page }) => {
  test.setTimeout(600_000);
  fs.mkdirSync(OUT, { recursive: true });
  const R = { pageErrors: [] };
  page.on('pageerror', (e) => R.pageErrors.push(e.message));

  await page.goto('/', { waitUntil: 'domcontentloaded' });
  await page.waitForFunction(() => !!window.__MW?.game, null, { timeout: 30_000 });
  await page.evaluate((save) => {
    localStorage.setItem('mathwarriors.save', JSON.stringify(save));
    const mgr = window.__MW.game.scene;
    mgr.getScenes(true).forEach((s) => mgr.stop(s.scene.key));
    mgr.start('OverworldScene', {});
  }, freshSave());
  await page.waitForFunction(() => window.__MW_OVERWORLD?.ready === true, null, { timeout: 120_000 });

  // ── DEFECT 9: does the first-arrival cinematic trigger on a fresh save? ──
  await page.waitForFunction(() => {
    const s = window.__MW.game.scene.getScene('OverworldScene');
    return s && !s._cover;
  }, null, { timeout: 60_000 });
  const cine = { samples: [] };
  for (let i = 0; i < 20; i++) {
    cine.samples.push(await page.evaluate(() => ({
      active: window.__MW_OVERWORLD.cinematicActive(),
      sim: window.__MW_OVERWORLD.stats().simTime,
    })));
    await page.waitForTimeout(400);
  }
  cine.everActive = cine.samples.some((s) => s.active);
  R.cinematic = cine;
  await page.screenshot({ path: `${OUT}/01-arrival.png` });

  // ── worldStats snapshot (creatures? colliders? hero rig) ──
  R.worldStats = await page.evaluate(() => {
    const w = window.__MW_OVERWORLD.worldStats();
    return { colliders: w.colliders, props: w.props, hero: w.hero };
  });

  // ── DEFECT 5: hero squash/stretch channels — idle, walk, jump-land ──
  const sample = () => page.evaluate(() => ({
    sim: window.__MW_OVERWORLD.stats().simTime,
    ...window.__MW_OVERWORLD.feel(),
    grounded: !!window.__MW_OVERWORLD._state.grounded,
    y: window.__MW_OVERWORLD._state.pos.y,
  }));
  // stop any cinematic and settle
  await page.evaluate(() => window.__MW_OVERWORLD.teleport(6, 158, Math.PI));
  await waitSim(page, 1.5);
  const idle = [];
  for (let i = 0; i < 10; i++) { idle.push(await sample()); await page.waitForTimeout(300); }
  R.heroIdle = idle;

  const walk = [];
  await page.keyboard.down('w');
  for (let i = 0; i < 10; i++) { walk.push(await sample()); await page.waitForTimeout(300); }
  await page.keyboard.up('w');
  R.heroWalk = walk;

  // jump + land
  await waitSim(page, 1.0);
  await page.keyboard.press('Space');
  const jump = [];
  for (let i = 0; i < 30; i++) { jump.push(await sample()); await page.waitForTimeout(120); }
  R.heroJump = jump;
  R.heroJumpMaxSquash = Math.max(...jump.map((s) => s.squash));
  // post-landing decay: does squash return to 0 and stay?
  await waitSim(page, 3);
  R.heroPostLand = await sample();

  // hero close-up screenshot (frozen pose, deterministic)
  await page.evaluate(() => window.__MW_OVERWORLD.setPose('hero-closeup'));
  await page.waitForTimeout(1500);
  await page.screenshot({ path: `${OUT}/02-hero-closeup.png` });
  await page.evaluate(() => window.__MW_OVERWORLD.freeze(false));

  // ── DEFECT 6: walk INTO a known tree ──
  // Nearest tree to spawn (deterministic, dumped from createProps in Node):
  // tree at (13.62, 155.96), trunk collider r=0.387. PLAYER_RADIUS=0.6.
  const TREE = { x: 13.62, z: 155.96, r: 0.387 };
  const start = { x: TREE.x - 3.0, z: TREE.z + 0.0 };
  const yaw = Math.atan2(TREE.x - start.x, TREE.z - start.z); // forward=(sin,cos)
  await page.evaluate(({ s, yaw }) => window.__MW_OVERWORLD.teleport(s.x, s.z, yaw), { s: start, yaw });
  await waitSim(page, 0.5);
  const treeRun = [];
  await page.keyboard.down('w');
  {
    const t0 = await simTime(page);
    while ((await simTime(page)) < t0 + 4) {
      const p = await pos(page);
      treeRun.push({ ...p, d: Math.hypot(p.x - TREE.x, p.z - TREE.z) });
      await page.waitForTimeout(150);
    }
  }
  await page.keyboard.up('w');
  R.treeCollision = {
    tree: TREE,
    minDistToTrunk: Math.min(...treeRun.map((s) => s.d)),
    expectedStopDist: TREE.r + 0.6,
    crossedToFarSide: treeRun.some((s) => s.x > TREE.x + 0.5),
    path: treeRun,
  };
  await page.screenshot({ path: `${OUT}/03-tree-walkthrough.png` });

  // control: building collider (shop at -162,-14 r=3.4) — does ANY prop block?
  const SHOP = { x: -162, z: -14, r: 3.4 };
  const bs = { x: SHOP.x, z: SHOP.z + 8 };
  const byaw = Math.atan2(SHOP.x - bs.x, SHOP.z - bs.z);
  await page.evaluate(({ s, yaw }) => window.__MW_OVERWORLD.teleport(s.x, s.z, yaw), { s: bs, yaw: byaw });
  await waitSim(page, 0.5);
  const shopRun = [];
  await page.keyboard.down('w');
  {
    const t0 = await simTime(page);
    while ((await simTime(page)) < t0 + 4) {
      const p = await pos(page);
      shopRun.push({ x: p.x, z: p.z, d: Math.hypot(p.x - SHOP.x, p.z - SHOP.z) });
      await page.waitForTimeout(150);
    }
  }
  await page.keyboard.up('w');
  R.buildingCollision = {
    shop: SHOP,
    minDist: Math.min(...shopRun.map((s) => s.d)),
    expectedStopDist: SHOP.r + 0.6,
    crossed: shopRun.some((s) => s.z < SHOP.z - 0.5),
  };

  // ── DEFECT 7: tree row close-up vs the 1.72 m hero ──
  // Stand hero next to trees near spawn garden: trees at (10.36,169.29) & (3.28,169.42)
  await page.evaluate(() => window.__MW_OVERWORLD.teleport(7.0, 163.5, Math.atan2(0.3, 5.8)));
  await waitSim(page, 1.0);
  await page.screenshot({ path: `${OUT}/04-tree-row.png` });

  // ── DEFECT 8: does water visibly animate? ──
  // Teleport to the tidepool shore (unfrozen sim, no input → camera drifts only
  // by tiny authored idle drift). Screenshot at t, t+5s, t+10s sim.
  await page.evaluate(() => window.__MW_OVERWORLD.teleport(155, 129, 0.87));
  await waitSim(page, 1.5);
  const shots = [];
  for (let i = 0; i < 3; i++) {
    const b = await page.screenshot();
    shots.push(b.toString('base64'));
    fs.writeFileSync(`${OUT}/05-water-t${i}.png`, b);
    if (i < 2) await waitSim(page, 5);
  }
  // diff in-page via 2D canvas: mean abs luma diff per region
  R.water = await page.evaluate(async (b64s) => {
    const imgs = [];
    for (const b of b64s) {
      const img = new Image();
      img.src = 'data:image/png;base64,' + b;
      await img.decode();
      imgs.push(img);
    }
    const W = imgs[0].width, H = imgs[0].height;
    const cv = document.createElement('canvas'); cv.width = W; cv.height = H;
    const ctx = cv.getContext('2d', { willReadFrequently: true });
    const data = imgs.map((im) => { ctx.clearRect(0, 0, W, H); ctx.drawImage(im, 0, 0); return ctx.getImageData(0, 0, W, H).data; });
    const region = (x0, y0, x1, y1, a, b) => {
      let sum = 0, n = 0;
      for (let y = y0; y < y1; y += 2) for (let x = x0; x < x1; x += 2) {
        const i = (y * W + x) * 4;
        sum += Math.abs(a[i] - b[i]) + Math.abs(a[i + 1] - b[i + 1]) + Math.abs(a[i + 2] - b[i + 2]);
        n++;
      }
      return +(sum / (n * 3)).toFixed(3);
    };
    // facing yaw 0.87 toward the sea: water fills the mid/right frame; sky top.
    const waterRect = [Math.floor(W * 0.55), Math.floor(H * 0.45), Math.floor(W * 0.95), Math.floor(H * 0.75)];
    const groundRect = [Math.floor(W * 0.05), Math.floor(H * 0.75), Math.floor(W * 0.35), Math.floor(H * 0.95)];
    const skyRect = [Math.floor(W * 0.1), Math.floor(H * 0.05), Math.floor(W * 0.9), Math.floor(H * 0.15)];
    return {
      water01: region(...waterRect, data[0], data[1]),
      water12: region(...waterRect, data[1], data[2]),
      ground01: region(...groundRect, data[0], data[1]),
      sky01: region(...skyRect, data[0], data[1]),
    };
  }, shots);

  console.log('FORENSIC_JSON_START');
  console.log(JSON.stringify(R, (k, v) => (k === 'path' ? v.map((p) => ({ x: +p.x.toFixed(2), z: +p.z.toFixed(2), d: +p.d.toFixed(2) })) : v), 1));
  console.log('FORENSIC_JSON_END');
});
