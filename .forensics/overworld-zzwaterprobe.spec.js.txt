import { test } from '@playwright/test';
import fs from 'node:fs';

const OUT = '.forensics';

function freshSave() {
  return {
    version: 1, grade: 3,
    party: [{ id: 'knight-shadow', name: 'Shadow', hp: 52, maxHp: 52, xp: 0, level: 1 }],
    gold: 10, potions: 2,
    floors: Array.from({ length: 9 }, (_, i) => ({ id: i + 1, unlocked: i < 3, complete: i < 2, bestStreak: 0 })),
    settings: { musicVolume: 0, sfxVolume: 0, reducedMotion: false },
    stats: { totalBattles: 1, totalCorrect: 0, totalWrong: 0, playTimeSec: 0, firstPlayedAt: Date.now(), lastPlayedAt: Date.now() },
  };
}

test.beforeEach(async ({ context }) => {
  await context.route(/^https?:\/\/(?!127\.0\.0\.1|localhost)/, (r) => r.abort());
});

async function boot(page) {
  await page.goto('/', { waitUntil: 'domcontentloaded' });
  await page.waitForFunction(() => !!window.__MW?.game, null, { timeout: 30_000 });
  await page.evaluate((save) => {
    localStorage.setItem('mathwarriors.save', JSON.stringify(save));
    const mgr = window.__MW.game.scene;
    mgr.getScenes(true).forEach((s) => mgr.stop(s.scene.key));
    mgr.start('OverworldScene', {});
  }, freshSave());
  await page.waitForFunction(() => window.__MW_OVERWORLD?.ready === true, null, { timeout: 120_000 });
  await page.waitForFunction(() => {
    const s = window.__MW.game.scene.getScene('OverworldScene');
    return s && !s._cover;
  }, null, { timeout: 60_000 });
}

async function diffProbe(page, tag, n = 5, gap = 1.5) {
  const shots = [];
  for (let i = 0; i < n; i++) {
    const b = await page.screenshot();
    shots.push(b.toString('base64'));
    fs.writeFileSync(`${OUT}/${tag}-t${i}.png`, b);
    const t0 = await page.evaluate(() => window.__MW_OVERWORLD.stats().simTime);
    await page.waitForFunction((t) => window.__MW_OVERWORLD.stats().simTime >= t, t0 + gap, { timeout: 120_000 });
  }
  return page.evaluate(async (b64s) => {
    const imgs = [];
    for (const b of b64s) {
      const img = new Image();
      img.src = 'data:image/png;base64,' + b;
      await img.decode();
      imgs.push(img);
    }
    const W = imgs[0].width, H = imgs[0].height;
    const cv = document.createElement('canvas'); cv.width = W; cv.height = H;
    const ctx = cv.getContext('2d', { willReadFrequently: true });
    const data = imgs.map((im) => { ctx.clearRect(0, 0, W, H); ctx.drawImage(im, 0, 0); return ctx.getImageData(0, 0, W, H).data; });
    const region = (x0, y0, x1, y1, a, b) => {
      let sum = 0, nn = 0;
      for (let y = y0; y < y1; y += 2) for (let x = x0; x < x1; x += 2) {
        const i = (y * W + x) * 4;
        sum += Math.abs(a[i] - b[i]) + Math.abs(a[i + 1] - b[i + 1]) + Math.abs(a[i + 2] - b[i + 2]);
        nn++;
      }
      return +(sum / (nn * 3)).toFixed(3);
    };
    const out = [];
    for (let i = 0; i + 1 < data.length; i++) {
      out.push({
        whole: region(0, 0, W, H, data[i], data[i + 1]),
        mid: region(0, Math.floor(H * 0.35), W, Math.floor(H * 0.65), data[i], data[i + 1]),
        lower: region(0, Math.floor(H * 0.65), W, H, data[i], data[i + 1]),
      });
    }
    return out;
  }, shots);
}

test('pond probe (spawn garden pool)', async ({ page }) => {
  test.setTimeout(600_000);
  fs.mkdirSync(OUT, { recursive: true });
  await boot(page);
  // garden-pool at (-8,154) r=7.5 — the first water a player sees from SPAWN (6,158).
  await page.evaluate(() => window.__MW_OVERWORLD.teleport(1, 158, Math.atan2(-9, -4)));
  const t0 = await page.evaluate(() => window.__MW_OVERWORLD.stats().simTime);
  await page.waitForFunction((t) => window.__MW_OVERWORLD.stats().simTime >= t, t0 + 4, { timeout: 120_000 });
  const diffs = await diffProbe(page, '07-pondprobe');
  console.log('PONDPROBE_JSON', JSON.stringify(diffs));
});

test('water probe', async ({ page }) => {
  test.setTimeout(600_000);
  fs.mkdirSync(OUT, { recursive: true });
  await page.goto('/', { waitUntil: 'domcontentloaded' });
  await page.waitForFunction(() => !!window.__MW?.game, null, { timeout: 30_000 });
  await page.evaluate((save) => {
    localStorage.setItem('mathwarriors.save', JSON.stringify(save));
    const mgr = window.__MW.game.scene;
    mgr.getScenes(true).forEach((s) => mgr.stop(s.scene.key));
    mgr.start('OverworldScene', {});
  }, freshSave());
  await page.waitForFunction(() => window.__MW_OVERWORLD?.ready === true, null, { timeout: 120_000 });
  await page.waitForFunction(() => {
    const s = window.__MW.game.scene.getScene('OverworldScene');
    return s && !s._cover;
  }, null, { timeout: 60_000 });

  // Face the sea with a spinning coin in frame: coin ow-tidepool-2 at (148,120).
  await page.evaluate(() => window.__MW_OVERWORLD.teleport(145, 123, Math.atan2(3, -3)));
  // settle camera fully
  await page.waitForFunction((t) => window.__MW_OVERWORLD.stats().simTime >= t,
    await page.evaluate(() => window.__MW_OVERWORLD.stats().simTime) + 4, { timeout: 120_000 });

  const shots = [];
  const meta = [];
  for (let i = 0; i < 5; i++) {
    meta.push(await page.evaluate(() => ({
      sim: window.__MW_OVERWORLD.stats().simTime,
      draws: window.__MW_OVERWORLD.stats().drawCalls,
    })));
    const b = await page.screenshot();
    shots.push(b.toString('base64'));
    fs.writeFileSync(`${OUT}/06-waterprobe-t${i}.png`, b);
    const t0 = await page.evaluate(() => window.__MW_OVERWORLD.stats().simTime);
    await page.waitForFunction((t) => window.__MW_OVERWORLD.stats().simTime >= t, t0 + 1.5, { timeout: 120_000 });
  }

  const diffs = await page.evaluate(async (b64s) => {
    const imgs = [];
    for (const b of b64s) {
      const img = new Image();
      img.src = 'data:image/png;base64,' + b;
      await img.decode();
      imgs.push(img);
    }
    const W = imgs[0].width, H = imgs[0].height;
    const cv = document.createElement('canvas'); cv.width = W; cv.height = H;
    const ctx = cv.getContext('2d', { willReadFrequently: true });
    const data = imgs.map((im) => { ctx.clearRect(0, 0, W, H); ctx.drawImage(im, 0, 0); return ctx.getImageData(0, 0, W, H).data; });
    const region = (x0, y0, x1, y1, a, b) => {
      let sum = 0, n = 0;
      for (let y = y0; y < y1; y += 2) for (let x = x0; x < x1; x += 2) {
        const i = (y * W + x) * 4;
        sum += Math.abs(a[i] - b[i]) + Math.abs(a[i + 1] - b[i + 1]) + Math.abs(a[i + 2] - b[i + 2]);
        n++;
      }
      return +(sum / (n * 3)).toFixed(3);
    };
    const whole = [0, 0, W, H];
    const upper = [0, Math.floor(H * 0.3), W, Math.floor(H * 0.6)]; // horizon band: open sea
    const lower = [0, Math.floor(H * 0.6), W, H];                   // near field: shore+coin+hero
    const out = [];
    for (let i = 0; i + 1 < data.length; i++) {
      out.push({
        whole: region(...whole, data[i], data[i + 1]),
        seaBand: region(...upper, data[i], data[i + 1]),
        nearField: region(...lower, data[i], data[i + 1]),
      });
    }
    return out;
  }, shots);

  console.log('WATERPROBE_JSON', JSON.stringify({ meta, diffs }));
});
